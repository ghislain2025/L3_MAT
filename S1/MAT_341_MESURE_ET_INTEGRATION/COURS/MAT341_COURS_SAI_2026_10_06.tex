\section*{Chapitre 01 : Anneaux et algèbres des parties d'un ensemble}

\begin{enonce}{Définition}{1}
On appelle \emph{anneau des parties d'un ensemble $E$} une famille $\A$ non vide de parties de $E$,
possédant les propriétés suivantes :
\begin{enumerate}[label=\textbf{\arabic*.},leftmargin=1.2cm]
  \item Si $A\in\A$ et $B\in\A$, alors $A\cup B\in\A$.
  \item Si $A\in\A$ et $B\in\A$, alors $A\setminus B\in\A$.
\end{enumerate}
\end{enonce}

\textbf{\color{bleu}Remarque 2.}
\begin{center}
\begin{tikzpicture}[scale=0.9]
  \fill[pattern=north east lines,pattern color=red!70!black] (0,0) circle (1);
  \fill[white] (1.2,0) circle (1);
  \draw[thick,bleu] (0,0) circle (1);
  \draw[thick,bleu] (1.2,0) circle (1);
  \node at (-1.25,0.9) {$A$};
  \node at (2.45,0.9) {$B$};
  \draw[->,red!70!black] (-0.9,-1.5) node[below,black]{$A\setminus B$} -- (-0.5,-0.4);
\end{tikzpicture}
\end{center}
\begin{center}
  \fcolorbox{red!70!black}{white}{\ \ $A\setminus B = A\cap B^{c}$\ \ }
\end{center}

\textbf{\color{bleu}Conséquence 3.}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item Soit $A\in\A$, alors $A\setminus A=\varnothing\in\A$, donc si $\A$ est un anneau, alors $\varnothing\in\A$.
  \item Soit $A\in\A$ et $B\in\A$, alors
  \[
    A\cap B=(A\cup B)\setminus\bigl((A\setminus B)\cup(B\setminus A)\bigr)\in\A .
  \]
\end{enumerate}
Ainsi, si $\A$ est un anneau et si $A\in\A$, $B\in\A$, alors $A\cap B\in\A$.

\textbf{\color{bleu}Remarque 4.}
$\A$ s'appelle un « anneau » car c'est un anneau au sens algébrique.
$\A$ est un anneau quand on le munit des opérations suivantes :
\begin{itemize}[leftmargin=1.2cm]
  \item une addition $\Delta$ qui est la \emph{différence symétrique} :
        $A\,\Delta\, B=(A\setminus B)\cup(B\setminus A)$ ;
  \item une multiplication $\cap$, qui est l'intersection.
\end{itemize}
$\Delta$ a bien les propriétés de l'addition dans l'anneau $\A$.

\begin{center}
\begin{tikzpicture}[scale=0.9]
  \draw[thick,bleu] (0,0) circle (1);
  \draw[thick,bleu] (1.2,0) circle (1);
  \draw[thick,bleu] (0.6,-1.0) circle (1);
  \node at (-1.25,0.9) {$A$};
  \node at (2.45,0.9) {$B$};
  \node at (0.6,-2.25) {$C$};
\end{tikzpicture}
\end{center}

En effet :
\begin{itemize}[leftmargin=1.2cm]
  \item $\Delta$ est commutative.
  \item $\A$ possède un élément neutre pour $\Delta$ ; en effet $\varnothing=A\,\Delta\,A$, pour tout $A\in\A$.
  \item $\Delta$ est associative : $(A\,\Delta\, B)\,\Delta\, C=A\,\Delta\,(B\,\Delta\, C)$.
\end{itemize}

\begin{note}
Pour être précis : $\varnothing\in\A$ est l'élément neutre ($A\,\Delta\,\varnothing=A$), et la relation
$A\,\Delta\,A=\varnothing$ signifie que chaque élément est son propre opposé. Le schéma à trois cercles
accompagne la vérification de l'associativité.
\end{note}

$\cap$ a bien les propriétés de la multiplication dans l'anneau :
\begin{itemize}[leftmargin=1.2cm]
  \item $\cap$ est bien associative, car $(A\cap B)\cap C=A\cap(B\cap C)$, $\ \forall A,B,C\in\A$.
  \item $\cap$ est bien distributive par rapport à $\Delta$, car
        $A\cap(B\,\Delta\, C)=(A\cap B)\,\Delta\,(A\cap C)$, $\ \forall A,B,C\in\A$.
\end{itemize}
On conclut alors que $(\A,\Delta,\cap)$ est un anneau au sens algébrique.

\textbf{\color{bleu}Exemple 5.}
\begin{enumerate}[label=\textbf{\arabic*-},leftmargin=1.2cm]
  \item Pour tout ensemble $E$, $\A=\Pp(E)$ est un anneau.
  \item Si $E=\R$, alors $\A$, l'ensemble des réunions finies d'intervalles finis, est un anneau.
  \begin{center}
  \begin{tikzpicture}[scale=0.9]
    \draw (0,0) -- (6,0);
    \foreach \x/\l in {0.8/a,2.2/c,3.8/b,5.2/d}{\draw (\x,0.08) -- (\x,-0.08); \node[below] at (\x,-0.1) {$\l$};}
  \end{tikzpicture}
  \end{center}
  Alors $[a;b]\setminus[c;d]=[a;c[$.
  \begin{center}
  \begin{tikzpicture}[scale=0.9]
    \draw (0,0) -- (6,0);
    \foreach \x/\l in {0.8/a,2.2/c,3.4/d,4.6/b}{\draw (\x,0.08) -- (\x,-0.08); \node[below] at (\x,-0.1) {$\l$};}
  \end{tikzpicture}
  \end{center}
  Alors $[a;b]\setminus[c;d]=[a;c[\,\cup\,]d;b]$.
  \item $\A$, l'ensemble des réunions finies d'intervalles $[a;b[$, fermés à gauche et ouverts à droite,
        est un anneau.
  \item $\A$, l'ensemble des réunions finies d'intervalles mixtes $]a;b]$, ouverts à gauche et fermés à droite,
        est un anneau.
\end{enumerate}

\begin{note}
Dans l'exemple 5.2, les deux cas du schéma sont $a<c<b<d$ puis $a<c<d<b$ ; j'ai ajouté ces hypothèses
d'ordre qui sont implicites sur les dessins.
\end{note}

\textbf{\color{bleu}Remarque 6.}
Si $\A$ est un anneau de parties de $E$ et si $E\in\A$, alors $\A$ a un élément neutre.

\begin{enonce}{Définition}{7}
On dit que la famille $\A$ des parties de $E$ est une \emph{algèbre} si $\A$ est un anneau et si $E\in\A$.
\end{enonce}

\textbf{\color{bleu}Remarque 8.}
Si $\A$ est une algèbre des parties de $E$, alors $\forall A\in\A$, $A^{c}\in\A$.
En effet $\bigl((E\in\A)\ \text{et}\ (A\in\A)\bigr)\Rightarrow A^{c}=E\setminus A\in\A$.
Les exemples précédemment cités (Exemple 5) ne sont pas des algèbres, car $\R$ n'est pas une réunion finie
d'intervalles finis.

\textbf{\color{bleu}Exemple 09.}
\begin{enumerate}[label=\textbf{\arabic*-},leftmargin=1.2cm]
  \item $\A$, l'ensemble des réunions finies d'intervalles finis ou infinis, est une algèbre.
  \item $\A$, l'ensemble des réunions finies d'intervalles mixtes $[a;b[$ ou $[a;+\infty[$ ou $]-\infty;b]$
        \ldots
\end{enumerate}

\begin{note}

\end{note}

\begin{enonce}{Définition}{10}
On appelle \emph{$\sigma$-anneau des parties d'un ensemble $E$} un anneau $\A$ qui est stable par réunions
dénombrables, c'est-à-dire qui possède la propriété suivante :
\[
  \forall A_m\in\A,\ m\in\N,\ \text{on a :}\quad \bigcup_{m\in\N}A_m\in\A .
\]
\end{enonce}

\begin{enonce}{Définition}{11}
On appelle \emph{$\sigma$-algèbre des parties d'un ensemble $E$} un $\sigma$-anneau $\A$ tel que $E\in\A$.
\end{enonce}

\textbf{\color{bleu}Remarque 12.}
Un ensemble dénombrable est un ensemble qui peut être mis en bijection avec $\N$ (ou une partie de $\N$).

\begin{enonce}{Propriété}{13}
Tout $\sigma$-anneau $\A$ est stable par intersection dénombrable, c'est-à-dire que pour toute famille
dénombrable $(A_m)_{m\in\N}\subset\A$, on a :
\[
  \bigcap_{n\in\N}A_n\in\A .
\]
\end{enonce}

\begin{preuve}\end{preuve}
\begin{itemize}[leftmargin=1.2cm]
  \item Si $E\in\A$ (c'est-à-dire que $\A$ est une $\sigma$-algèbre), alors
  \[
    \bigcap_{n\in\N}A_n=\Bigl(\bigcup_{n\in\N}A_n^{c}\Bigr)^{c}\in\A .
  \]
  \item Si $E\notin\A$, posons $B=\displaystyle\bigcup_{n\in\N}A_n\in\A$ et on a :
  \[
    \bigcap_{n\in\N}A_n=B\setminus\Bigl(\bigcup_{n\in\N}(B\setminus A_n)\Bigr).
  \]
\end{itemize}

\textbf{\color{bleu}Exemple 14.}
Si $E$ est un ensemble, alors $\Pp(E)$ est un $\sigma$-anneau, et c'est même une $\sigma$-algèbre.

\textbf{\color{bleu}Remarque 15.}
On utilise aussi les noms de « tribu » pour $\sigma$-algèbre et « clan » pour $\sigma$-anneau.

\begin{note}
Le mot « clan » est difficile à lire dans le manuscrit ; c'est le terme usuel pour $\sigma$-anneau.
\end{note}

\begin{enonce}{Proposition}{16}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item Si $(\A_i)_{i\in I}$ est une famille d'anneaux de parties de $E$, alors $\bigcap_{i\in I}\A_i$ est un anneau.
  \item Si $(\A_i)_{i\in I}$ est une famille d'algèbres des parties de $E$, alors $\bigcap_{i\in I}\A_i$ est une algèbre.

\begin{enonce}{Théorème et définition}{17}
\begin{enumerate}[label=\textbf{\arabic*.},leftmargin=1.2cm]
  \item Soit $\Ee$ une famille de parties d'un ensemble $E$ ($\Ee\subset\Pp(E)$). Parmi les anneaux contenant $\Ee$,
        il existe un plus petit anneau appelé « \emph{anneau engendré par $\Ee$} ».
  \item Parmi les $\sigma$-anneaux contenant $\Ee$, il existe un plus petit $\sigma$-anneau, appelé
        « \emph{$\sigma$-anneau engendré par $\Ee$} ».
  \item Parmi les tribus contenant $\Ee$, il existe une plus petite tribu appelée « \emph{tribu engendrée} ».
\end{enumerate}
\end{enonce}

qui contient $\Ee$ et $\A$ est contenue dans toute tribu qui contient $\Ee$.

\begin{enonce}{Définition}{18}
On appelle \emph{semi-anneau des parties de l'ensemble $E$} une famille $\Ss$ de parties de $E$ telle que :
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item $\forall A,B\in\Ss$, $A\cap B\in\Ss$ (stabilité par l'intersection) ;
  \item $\forall A,B\in\Ss$, il existe une famille finie d'éléments de $\Ss$ deux à deux disjoints
        $(C_i)_{i=1,\dots,p}$ telle que
        \[
          A\setminus B=\bigcup_{i=1}^{p}C_i .
        \]
\end{enumerate}
\end{enonce}

\textbf{\color{bleu}Exemple 19.} Si $E=\R$ :
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item l'ensemble $\Ii$ des intervalles est un semi-anneau ;
  \item l'ensemble $\Ii_g$ des intervalles finis de la forme $[a;b[$ est un semi-anneau ;
  \item l'ensemble $\Ii_d$ des intervalles finis de la forme $]a;b]$ est un semi-anneau.
\end{enumerate}
Dans $E=\R^n$ :
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item Les pavés forment un semi-anneau.
\end{enumerate}

\begin{note}
Le manuscrit écrit « un anneau » à l'exemple 19 pour les pavés de $\R^n$ ; le résultat correct (confirmé
par la remarque suivante) est qu'ils forment un \emph{semi-anneau}. J'ai aussi omis la mention « dans $E=\R^n$ »
qui apparaît en fin de la ligne 3).
\end{note}

\textbf{\color{bleu}Remarque.}
Un pavé de $\R^n$ est une partie $I_n$ de la forme
\[
  I_n=[a_1,b_1]\times[a_2,b_2]\times\cdots\times[a_n;b_n]
\]
(avec $a_i,b_i\in\R$, $\forall\,1\le i\le n$, $a_i\le b_i$).
Les pavés de la forme $[a_1;b_1[\times[a_2;b_2[\times\cdots\times[a_n;b_n[$ forment un semi-anneau.

\begin{enonce}{Proposition}{20}
L'anneau engendré par le semi-anneau $\Ss$ est formé de réunions finies des éléments de $\Ss$.
\end{enonce}

\begin{preuve}\end{preuve}
Tout anneau qui contient $\Ss$ contient les réunions finies d'éléments de $\Ss$.
Désignons par $\A$ l'ensemble des réunions finies d'éléments de $\Ss$. Alors tout anneau qui contient $\Ss$
va contenir $\A$. Il reste à démontrer que $\A$ est un anneau.

\textbf{1) Stabilité de $\A$ par réunion de deux éléments.}
Soit $A,B\in\A$, alors $A$ et $B$ sont des réunions finies d'éléments de $\Ss$, donc $A\cup B$ est aussi
une réunion finie d'éléments de $\Ss$.

\textbf{2) Stabilité par différence.} Soit $A,B\in\A$, alors
\[
  A=S_1\cup S_2\cup\cdots\cup S_p,\qquad B=S'_1\cup S'_2\cup\cdots\cup S'_q .
\]
Donc
\[
  A\setminus B=\Bigl(\bigl((A\setminus S'_1)\setminus S'_2\bigr)\setminus\cdots\Bigr)\setminus S'_q .
\]
Il suffit alors de montrer que $\forall A\in\A$, $S\in\Ss$, $A\setminus S\in\A$.
Mais
\[
  A\setminus S=(S_1\setminus S)\cup(S_2\setminus S)\cup\cdots\cup(S_p\setminus S).
\]
Or, par hypothèse, chaque $S_1\setminus S$, $S_2\setminus S$, \ldots{} est une réunion finie d'éléments de $\Ss$.
Donc $A\setminus S$ est aussi une réunion finie d'éléments de $\Ss$.



\begin{enonce}{Définition}{21}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item On appelle \emph{$\sigma$-algèbre des boréliens} la $\sigma$-algèbre engendrée par l'ensemble $\Oo$ d'ouverts
        de $\R$ ou $\R^n$. On la note $\B_{\R}$ ou $\B_{\R^n}$.
  \item Plus généralement, dans tout espace topologique, on appelle \emph{$\sigma$-algèbre de Boréliens}
        la $\sigma$-algèbre engendrée par l'ensemble des ouverts de l'espace.
  \item Soit $\B$ l'ensemble des boréliens de l'espace topologique $E$, et soit $\A$ la $\sigma$-algèbre
        engendrée par l'ensemble $\Ff$ des fermés de $E$.
\end{enumerate}
\end{enonce}

Montrons que $\A=\B$.

\textbf{i) Montrons que $\B\supset\A$.} Il suffit de montrer que $\B\supset\Ff$.
Or $\B$ est une tribu, donc elle contient tous les ouverts de $E$, c'est-à-dire qu'elle contient $\Oo$,
et comme les complémentaires des éléments de $\Oo$ font partie de $\B$, alors $\B\supset\Ff$.
Il vient que $\B\supset\A$.

\textbf{ii) Montrons que $\A$ contient $\B$ ($\A\supset\B$).} Il suffit de montrer que $\A$ contient $\Oo$.
Mais $\A\supset\Ff$ et les complémentaires des éléments de $\Ff$ appartiennent aussi à $\A$. Donc $\A\supset\Oo$.
Il en résulte que $\A\supset\B$.

\textbf{\color{bleu}Remarque.} Si $\A$ est la tribu engendrée par une famille $\Ee$, on peut noter $\A=\sigma(\Ee)$.

\begin{note}
$\Oo$ désigne l'ensemble des ouverts de $E$ (cerclé dans le manuscrit).
\end{note}



\begin{enonce}{Proposition}{22}
La tribu borélienne de $\R$, $\B_{\R}$, est aussi la tribu engendrée par l'ensemble $\Ii$ des intervalles.
\end{enonce}

\begin{preuve}\end{preuve}
Appelons $\A$ la tribu engendrée par $\Ii$. $\B_{\R}$ contient les intervalles ouverts qui sont des ouverts
et les intervalles fermés.

De plus, $]a,b]=\,]a,b[\,\cup\{b\}$ et $\{b\}$ est fermé, donc appartient à $\B_{\R}$. Alors $]a,b]\in\B_{\R}$.
De même, $[a,b[\,=\,]a,b[\,\cup\{a\}$, donc $[a,b[\,\in\B_{\R}$.
Par conséquent $\B_{\R}\supset\Ii$, d'où $\B_{\R}\supset\A$.

Montrons que $\B_{\R}\subset\A$. Il suffira de montrer que $\A$ contient les ouverts de $\R$.
Or sur $\R$ tout ouvert est une réunion dénombrable d'intervalles ouverts
(en effet les intervalles $]a;b[$, $a,b\in\Q$, forment une base dénombrable des ouverts de $\R$).
Donc $\B_{\R}\subset\A$.

\section*{Chapitre 2 : Espaces mesurables, applications mesurables}
\subsection*{2-1 Généralités}

\begin{enonce}{Définition}{1}
On appelle \emph{espace mesurable} $(E,\B)$ où $E$ est un ensemble et $\B$ une tribu des parties de $E$.
\end{enonce}

\textbf{\color{bleu}Exemple 2.} $(\R,\B_{\R})$ est un espace mesurable.

\begin{enonce}{Définition}{3}
Soient $(E,\A)$ et $(F,\B)$ deux espaces mesurables : une application $f:E\to F$ est dite \emph{mesurable}
si pour tout $A\in\B$, $f^{-1}(A)\in\A$.
\end{enonce}

Les éléments d'une tribu s'appellent des \emph{mesurables}.

\textbf{\color{bleu}Remarque.}
$f$ est mesurable si l'image réciproque par $f$ d'un mesurable de $F$ est un mesurable de $E$.

\begin{note}
La phrase « les éléments d'une tribu s'appellent des mesurables » est écrite en haut de la page suivante
du cahier ; je la place juste après la définition 3.
\end{note}

\begin{enonce}{Définition}{4}
Une application $f:\R\to\R$ est dite \emph{mesurable} si elle est mesurable de $(\R,\B_{\R})$ vers $(\R,\B_{\R})$.
\end{enonce}

\begin{enonce}{Théorème}{5}
Soit $E$ un ensemble, $(F,\B)$ un espace mesurable et $f:E\to F$, alors :
\begin{enumerate}[label=\textbf{\arabic*-},leftmargin=1.2cm]
  \item $f^{-1}(\B)=\bigl\{f^{-1}(A),\ A\in\B\bigr\}$ est une tribu des parties de $E$.
  \item Si $\Qq$ est une famille des parties de $F$ qui engendre la tribu $\B$ (i.e.\ $\B=\sigma(\Qq)$),
        alors $f^{-1}(\Qq)$ engendre la tribu $f^{-1}(\B)$.
\end{enumerate}
\end{enonce}

\begin{preuve}\end{preuve}
\textbf{Preuve du point 1.}

$\ast$ \emph{Stabilité par réunions dénombrables.} Soit $(A_n)_{n\in\N}\subset f^{-1}(\B)$, alors $\forall n\in\N$
il existe $B_n\in\B$ tel que $A_n=f^{-1}(B_n)$. Ainsi
\[
  \bigcup_{n\in\N}A_n=\bigcup_{n\in\N}f^{-1}(B_n)=f^{-1}\Bigl(\bigcup_{n\in\N}B_n\Bigr)\in f^{-1}(\B)
\]
car $\bigcup_{n\in\N}B_n\in\B$.

$\ast$ \emph{Stabilité par la différence.} Soit $A_1=f^{-1}(B_1)$, $A_2=f^{-1}(B_2)$, $B_1,B_2\in\B$, alors
\[
  A_1\setminus A_2=f^{-1}(B_1)\setminus f^{-1}(B_2)=f^{-1}(B_1\setminus B_2)\in f^{-1}(\B)
\]
car $B_1\setminus B_2\in\B$.

$\ast$ On sait que $E=f^{-1}(F)$ et $F\in\B$, donc $E\in f^{-1}(\B)$.

\medskip
\textbf{Preuve du point 2.}

Notons $\A$ la tribu engendrée par $f^{-1}(\Qq)$, et montrons que $\A=f^{-1}(\B)$.
Puisque $\B\supset\Qq$, alors $f^{-1}(\B)\supset f^{-1}(\Qq)$, d'où $f^{-1}(\B)\supset\A$,
la tribu engendrée par $f^{-1}(\Qq)$.

Il reste à montrer que $f^{-1}(\B)\subset\A$. Posons
\[
  \M=\bigl\{M\in\B \ \big/\ f^{-1}(M)\in\A\bigr\}\qquad(\text{par définition } \M\subset\B).
\]
Mais nous voulons démontrer que $\M=\B$, ce qui reviendra à conclure que $\B\subset\M$.

$\bullet$ Montrons d'abord que $\M$ est une tribu.

i) Soit $M_n\in\M$, $\forall n\in\N$. Alors $M_n\in\B$ et $f^{-1}(M_n)\in\A$ $\ \forall n\in\N$.
Comme $\bigcup_{n\in\N}M_n\in\B$ (car $\B$ est une tribu), alors
\[
  f^{-1}\Bigl(\bigcup_{n\in\N}M_n\Bigr)=\bigcup_{n\in\N}f^{-1}(M_n)\in\A\qquad(\text{car }\A\text{ est une tribu}),
\]
ainsi $\bigcup_{n\in\N}M_n\in\M$.

ii) Soient $M_1$ et $M_2\in\M$ alors $M_1,M_2\in\B$ et $f^{-1}(M_1),f^{-1}(M_2)\in\A$, donc
$M_1\setminus M_2\in\B$ et
\[
  f^{-1}(M_1\setminus M_2)=f^{-1}(M_1)\setminus f^{-1}(M_2)\in\A .
\]
Ainsi $M_1\setminus M_2\in\M$.

iii) On sait que $F\in\B$ et $f^{-1}(F)=E\in\A$, ainsi $F\in\M$.

$\ast$ Il reste à voir que $\B\subset\M$. Or $\A=\sigma\bigl(f^{-1}(\Qq)\bigr)\supset f^{-1}(\Qq)$.
Donc $\M\supset\Qq$. D'où $\M\supset\B$, la tribu engendrée par $\Qq$.

\begin{enonce}{Théorème}{6}
Soient $(E,\A)$, $(F,\B)$ deux espaces mesurables. Soit $\Qq$ une famille de parties qui engendre $\B$.
Soit $f:E\to F$ une application. Pour que $f$ soit mesurable il faut et il suffit que $f^{-1}(\Qq)\subset\A$.
\end{enonce}

\begin{preuve}\end{preuve}
\textbf{1- Hypothèse :} $f$ est mesurable de $(E,\A)$ vers $(F,\B)$. Comme $f$ est mesurable, $f^{-1}(\B)\subset\A$,
or $\Qq\subset\B$, donc $f^{-1}(\Qq)\subset\A$.

\textbf{2- Hypothèse :} $f^{-1}(\Qq)\subset\A$, donc la tribu $\A$ contient aussi la tribu engendrée par
$f^{-1}(\Qq)$, donc $\A\supset f^{-1}(\B)$ d'après le théorème 5.

\begin{tcolorbox}[colback=gray!8,colframe=gray!60,boxrule=0pt,leftrule=2pt,arc=0pt,left=6pt,top=2pt,bottom=2pt]
\textbf{Note :} Toute application $f:(\R,\B_{\R})\to(\R,\B_{\R})$ qui est continue est mesurable, car si $O$ est un ouvert de $\R$,
alors $f^{-1}(O)$ est un ouvert. Donc c'est un élément de $\B_{\R}$ et ceci est suffisant, car l'ensemble des ouverts
engendre $\B_{\R}$ (Théorème 6).
\end{tcolorbox}

\begin{enonce}{Application}{7}
Si $E$ et $F$ sont des espaces topologiques, et si $\A$ est la tribu des boréliens de $E$ et $\B$ la tribu des
boréliens de $F$, alors toute application $f:E\to F$ continue est une application mesurable de $(E,\A)\to(F,\B)$.
\end{enonce}

\begin{preuve}\end{preuve}
Par définition des boréliens de $F$, l'ensemble $\Oo$ des ouverts de $F$ engendre $\B$.
Pour montrer que $f$ est mesurable il suffit de montrer que $f^{-1}(\Oo)\subset\A$.
Or $f$ étant continue, $\forall A\in\Oo$, $f^{-1}(A)$ est un ouvert de $E$, donc $f^{-1}(A)\in\A$.

Soit $f:(\R,\B_{\R})\to(\R,\B_{\R})$. Puisque la famille $]-\infty,a]$ $(a\in\R)$ engendre $\B_{\R}$,
pour montrer que $f$ est mesurable, il suffit de montrer que
\[
  f^{-1}\bigl(]-\infty,a]\bigr)\in\B_{\R}\qquad\forall a\in\R .
\]

\begin{enonce}{Théorème}{8 \textnormal{(Fonctions composées mesurables)}}
Soient $(E,\A)$, $(F,\B)$, $(G,\C)$ trois espaces mesurables. Soient $f:E\to F$, $g:F\to G$ mesurables, alors
l'application $g\circ f:(E,\A)\to(G,\C)$ est mesurable.
\end{enonce}

\begin{note}
Le numéro du théorème est raturé sur le cahier ; je lis « 8 » d'après la suite de la numérotation (le prochain
numéro écrit est 10).
\end{note}

\begin{preuve}\end{preuve}
Soit $C\in\C$, montrons que $(g\circ f)^{-1}(C)\in\A$. Or
\[
  (g\circ f)^{-1}(C)=f^{-1}\bigl(g^{-1}(C)\bigr).
\]
Comme $g$ est mesurable, alors $g^{-1}(C)\in\B$ et $f$ étant mesurable, $f^{-1}\bigl(g^{-1}(C)\bigr)\in\A$,
d'où $(g\circ f)^{-1}(C)\in\A$.

\begin{note}
\textbf{Numéro 9 reconstitué} d'après le cours de F.~Wamon (MAT~303), Remarque 3.1 ; absent du manuscrit.
\end{note}

\begin{enonce}{Remarque}{9 \textnormal{(reconstituée)}}
Si $E$ est un espace topologique, sauf mention du contraire, $E$ sera muni de la tribu $\B_E$ des boréliens de $E$.

Soit $F$ un espace topologique et $(E,\A)$ un espace mesurable. Une application $f$ de $E$ dans $F$ est dite
mesurable si $f$ est mesurable de $(E,\A)$ dans $(F,\B_F)$. En particulier,
\[
  f:E\to\R \text{ mesurable}\iff f:(E,\A)\to(\R,\B_{\R}) \text{ mesurable}\iff \forall A\in\B_{\R},\ f^{-1}(A)\in\A .
\]
\end{enonce}

\subsection*{2-2 Fonctions numériques mesurables}
\subsubsection*{L'ensemble $\Rb$}

\begin{enonce}{Définition}{10}
L'ensemble $\Rb$ est l'ensemble $\R$ auquel on ajoute deux éléments notés $+\infty$ et $-\infty$.
Donc $\Rb=\R\cup\{+\infty,-\infty\}$.

$\Rb$ est muni d'une relation d'ordre totale ainsi définie :
$\forall x,y\in\R$, $x\le y$ dans $\Rb\iff x\le y$ dans $\R$ ;
$\forall y\in\Rb$, $-\infty\le y$ ; $\ \forall x\in\Rb$, $x\le+\infty$.
\end{enonce}

\begin{enonce}{Conséquence}{11}
$\Rb$ a un plus grand élément $(+\infty)$ et un plus petit élément $(-\infty)$.
\end{enonce}

\begin{enonce}{Propriété}{12}
Dans $\Rb$, toute partie $A$ admet une borne supérieure et une borne inférieure.
\end{enonce}

\begin{preuve}\end{preuve}
Soit $A\subset\Rb$.
\begin{enumerate}[label=\textbf{\arabic*-},leftmargin=1.2cm]
  \item Si $A$ est majorée dans $\R$, $A$ admet une borne sup dans $\R$ qui l'est dans $\Rb$.
  \item Si $A\subset\Rb$ et non majorée dans $\R$, puisque $A$ est non majorée dans $\R$, alors $A$ n'admet qu'un seul
        majorant dans $\Rb$, et sa borne sup est $+\infty$.
  \item Si $A\not\subset\R$ :
  \begin{itemize}[leftmargin=0.8cm]
    \item si $+\infty\in A$, $+\infty$ est l'unique majorant, donc la borne sup ;
    \item si $-\infty\in A$ et $+\infty\notin A$ :
    \begin{itemize}[leftmargin=0.8cm]
      \item si $A\cap\R\neq\varnothing$, $A$ a la même borne sup que $A\cap\R$ ;
      \item si $A\cap\R=\varnothing$, $A=\{-\infty\}$, donc sa borne sup est $-\infty$.
    \end{itemize}
  \end{itemize}
\end{enumerate}



\begin{note}
Seule la borne supérieure est traitée dans le manuscrit ; la borne inférieure se montre de la même façon.
\end{note}

\begin{enonce}{Notation}{13}
$\sup A$ désigne la borne supérieure de $A$ ; $\ \inf A$ désigne la borne inférieure de $A$.
\end{enonce}

\textbf{Topologie de $\Rb$} \textit{(suite des définitions de $\Rb$)}.
La famille des intervalles ouverts de $\Rb$ : $]a,b[$, $[-\infty,a[$, $]a;+\infty]$ est une base d'ouverts
définissant la topologie de $\Rb$.

\begin{enonce}{Conséquence}{14}
$\forall x\in\Rb$, les intervalles ouverts contenant $x$ forment une base de voisinages dans $\Rb$
comme dans $\R$.
\end{enonce}

\textbf{\color{bleu}Remarque.}
\begin{itemize}[leftmargin=1.2cm]
  \item Si $a=-\infty$, les $[-\infty,a[$ forment une base de voisinages de $-\infty$.
  \item Si $a=+\infty$, les $]a,+\infty]$ forment une base de voisinages de $+\infty$.
  \item Tout ouvert de $\Rb$ est une réunion dénombrable d'intervalles ouverts.
\end{itemize}

\begin{enonce}{Propriété}{15}
\begin{enumerate}[label=\textbf{\arabic*.},leftmargin=1.2cm]
  \item Dans $\Rb$ toute suite monotone est convergente (faux dans $\R$).
  \item L'ensemble des boréliens de $\Rb$, $\B_{\Rb}$, se déduit de $\B_{\R}$ ainsi : $\forall A\in\B_{\Rb}$,
        ou bien $A\in\B_{\R}$, ou $A=B\cup\{+\infty\}$, $B\in\B_{\R}$, ou $A=B\cup\{-\infty\}$, $B\in\B_{\R}$,
        ou $A=B\cup\{-\infty,+\infty\}$, $B\in\B_{\R}$.
\end{enumerate}
\end{enonce}

\begin{enonce}{Conséquence}{16}
De la topologie de $\Rb$, on déduit la topologie de $\Rb\times\Rb$.
On a une base ouverte de $\Rb\times\Rb$ en prenant les pavés définis par les produits de deux intervalles ouverts de $\Rb$.

L'ensemble des boréliens de $\Rb\times\Rb$, $\B_{\Rb\times\Rb}$, se déduit de $\B_{\R\times\R}$ ainsi :
si $A\in\B_{\Rb\times\Rb}$, alors ou bien $A\in\B_{\R\times\R}$, ou $A=B\cup C$ où $B\in\B_{\R\times\R}$ et
\[
  C=\{+\infty\}\times K_1\ \cup\ \{-\infty\}\times K_2\ \cup\ K_3\times\{-\infty\}\ \cup\ K_4\times\{+\infty\},
\]
où $K_1,K_2,K_3,K_4\in\B_{\R}$.
\end{enonce}

\begin{note}
Les quatre points $(\pm\infty,\pm\infty)$ ne sont couverts par cette écriture que si l'on autorise $K_i\in\B_{\Rb}$ ; le manuscrit écrit $K_i\in\B_{\R}$.
\end{note}

\section*{Chapitre 2 (suite)}
\subsubsection*{Opérations sur $\Rb$ (fin de la définition de $\Rb$)}

$\Rb$ est muni de deux opérations ainsi définies :

\textbf{a) L'addition.} L'addition est commutative et :
\begin{itemize}[leftmargin=1.2cm]
  \item $\forall x,y\in\R$, $x+y$ a la même valeur dans $\Rb$ que dans $\R$.
  \item $\forall x\in\R$, $x+(+\infty)=+\infty$, $\ x+(-\infty)=-\infty$,\\
        $(+\infty)+(+\infty)=+\infty$, $\ (-\infty)+(-\infty)=-\infty$.
\end{itemize}
L'addition est donc définie seulement sur la partie
$\Sigma=\Rb\times\Rb\setminus\bigl\{(-\infty,+\infty),(+\infty,-\infty)\bigr\}$ de $\Rb\times\Rb$.

\begin{enonce}{Conséquence}{17}
L'addition est associative.
\end{enonce}

\textbf{b) La multiplication.} La multiplication dans $\Rb$ est commutative et ainsi définie :
\begin{itemize}[leftmargin=1.2cm]
  \item $\forall x,y\in\R$, $x\cdot y$ a la même valeur dans $\Rb$ que dans $\R$.
  \item $\forall x\in\R$, $\ x\cdot(+\infty)=\begin{cases}+\infty&\text{si }x>0\\-\infty&\text{si }x<0\end{cases}$
  \item $\forall x\in\R$, $\ x\cdot(-\infty)=\begin{cases}-\infty&\text{si }x>0\\+\infty&\text{si }x<0\end{cases}$
  \item $(+\infty)\cdot(+\infty)=+\infty$, $\ (+\infty)\cdot(-\infty)=-\infty$, $\ (-\infty)\cdot(-\infty)=+\infty$ ;
        $\ 0\cdot(-\infty)=0$, $\ 0\cdot(+\infty)=0$.
\end{itemize}

\begin{enonce}{Conséquence}{18}
\begin{enumerate}[label=\textbf{\roman*)},leftmargin=1.2cm]
  \item La multiplication dans $\Rb$ est associative : $\forall x,y,z\in\Rb$, $(x\cdot y)\cdot z=x\cdot(y\cdot z)$.
  \item L'addition est une opération continue de $\Sigma=\Rb\times\Rb\setminus\{(+\infty,-\infty),(-\infty,+\infty)\}$ dans $\Rb$.
        D'où elle est mesurable.
  \item La multiplication n'est pas continue aux points $(+\infty,0)$, $(0,+\infty)$, $(-\infty,0)$, $(0,-\infty)$, mais elle est
        mesurable de $(\Rb\times\Rb,\B_{\Rb\times\Rb})$ dans $(\Rb,\B_{\Rb})$.
\end{enumerate}
\end{enonce}

\begin{note}
Dans le manuscrit, le cas $x\cdot(-\infty)$ pour $x>0$ est écrit avec un signe peu lisible (« $x\ge0$ » raturé) ; je lis « $x>0$ »,
comme dans la feuille de cours du 29-09-26.
\end{note}

\subsubsection*{Limites supérieures, limites inférieures, limite}

Soit $(x_n)_{n\in\N}\subset\Rb$ une suite. Alors $A_n=\{x_n,x_{n+1},\dots\}$ est une partie non vide (et bornée dans $\Rb$ par $+\infty$ et $-\infty$).
Donc $\sup(A_n)=\sup_{p\ge n}x_p$ et $\inf(A_n)=\inf_{p\ge n}x_p$ existent.

\textbf{\color{bleu}Remarque.} Comme $A_{n+1}\subset A_n$, alors $\sup A_{n+1}\le\sup A_n$ et $\inf A_{n+1}\ge\inf A_n$.

Donc $\sup_{p\ge n}x_p$ est décroissante dans $\Rb$ et minorée, donc converge vers sa borne inférieure ; ainsi
\[
  \lim_{n\to+\infty}\ \sup_{p\ge n}x_p=\inf_{n\in\N}\ \sup_{p\ge n}x_p=\overline{\lim}\,x_n .
\]
De même, $\inf_{p\ge n}x_p$ est croissante et majorée dans $\Rb$, donc converge vers sa borne supérieure ; ainsi
\[
  \lim_{n\to+\infty}\ \inf_{p\ge n}x_p=\sup_{n\in\N}\ \inf_{p\ge n}x_p=\underline{\lim}\,x_n .
\]
On observe aussi que
\[
  \inf_{p\ge n}x_p\ \le\ x_n\ \le\ \sup_{p\ge n}x_p .
\]
Donc
\[
  \lim_{n\to+\infty}\ \inf_{p\ge n}x_p\ \le\ \lim x_n\ \le\ \lim_{n\to+\infty}\ \sup_{p\ge n}x_p ,
\]
il vient que $\underline{\lim}\,x_n\le\lim x_n\le\overline{\lim}\,x_n$.

\begin{enonce}{Définition}{19}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item On appelle \emph{limite supérieure} de la suite $(x_n)_{n\in\N}$ dans $\Rb$ le nombre ainsi défini :
  \[
    \limsup x_n=\lim_{n\to+\infty}\ \sup_{p\ge n}x_p=\inf_{n\in\N}\ \sup_{p\ge n}x_p .
  \]
  On écrit aussi $\overline{\lim}\,x_n$ et on dit « plus grande limite ».
  \item On appelle \emph{limite inférieure} de la suite $(x_n)_{n\in\N}$ dans $\Rb$ le nombre ainsi défini :
  \[
    \liminf x_n=\lim_{n\to+\infty}\ \inf_{p\ge n}x_p=\sup_{n\in\N}\ \inf_{p\ge n}x_p .
  \]
  On écrit aussi $\underline{\lim}\,x_n$ et on dit aussi « plus petite limite ».
\end{enumerate}
\end{enonce}

\begin{enonce}{Proposition}{20}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item Pour que $\ell$ soit la limite supérieure de la suite $(x_n)_{n\in\N}$, il faut et il suffit que :
  \begin{itemize}[leftmargin=0.8cm]
    \item $\forall z>\ell$, il y a un nombre fini d'indices $n$ tels que $x_n>z$ ;
    \item $\forall z<\ell$, il y a un nombre fini d'indices $n$ tels que $x_n>z$.
  \end{itemize}
  \item Pour que $m$ soit la limite inférieure de la suite $(x_n)_{n\in\N}$, il faut et il suffit que :
  \begin{itemize}[leftmargin=0.8cm]
    \item $\forall z<m$, il y a un nombre fini d'indices $n$ tels que $x_n<z$ ;
    \item $\forall z>m$, il y a une infinité d'indices $n$ tels que $x_n<z$.
  \end{itemize}
\end{enumerate}
\end{enonce}

\begin{note}
Dans le manuscrit, la deuxième condition du point 1) est écrite « il y a un nombre fini d'indices » ; la preuve (nécessité) conclut
à « une infinité » et la feuille de cours du 29-09-26 écrit « une infinité » : c'est sans doute un lapsus. Je transcris le manuscrit tel quel.
\end{note}

\begin{preuve}\end{preuve}
\textbf{1) Nécessité.} Supposons que $\ell=\limsup x_n=\inf_{n\in\N}\sup_{p\ge n}x_p$. Posons $y_n=\sup_{p\ge n}x_p$, alors
$\ell=\inf_{n\in\N}y_n$, car la suite $(y_n)$ est décroissante et converge vers $\ell$.

Donc $\forall z>\ell$, $\exists n_0\in\N$ tel que $y_{n_0}<z$, d'où $\sup_{p\ge n_0}x_p<z$.

\emph{Rappel :} $\theta=\inf A\iff\begin{cases}\forall x\in A,\ \theta\le x\\ \forall\varepsilon>0,\ \exists x_0\in A,\ x_0<\theta+\varepsilon.\end{cases}$
Posons $\varepsilon=z-\ell>0$, $\exists n_0\in\N$ tel que $y_{n_0}<\ell+(z-\ell)$.

Donc si $x_p>z$ alors $p<n_0$ ; d'où il y a un nombre fini d'indices.

Si $z<\ell$ alors $\forall n\in\N$, $z\le y_n$. Donc $\forall n\in\N$, $z\le\sup_{p\ge n}x_p$ ; ainsi il existe $p_0\ge n$ tel que $z<x_{p_0}$.
Comme c'est vrai pour tout $n\in\N$, alors il existe une infinité de $x_{p_0}>z$.

\textbf{Suffisance.} $\ell$ est un nombre tel que $\forall z>\ell$ il y a un nombre fini d'indices $n$ tels que $x_n>z$ et $\forall z>\ell$
il y a une infinité d'indices $n$ tels que $x_n>z$.

\begin{note}
Dans la suffisance, le manuscrit écrit « $\forall z>\ell$ » pour la seconde condition ; d'après la suite de la preuve, il faut lire « $\forall z<\ell$ ».
\end{note}

On veut montrer que $\ell=\limsup x_n$ : posons $\ell'=\limsup x_n$ et montrons que $\ell=\ell'$.

Soit $z>\ell$ et $n_0$ le plus grand indice tel que $x_n>z$, donc pour $n>n_0$, $x_n\le z$. Ainsi $y_n=\sup_{p\ge n}x_p\le z$ pour $n>n_0$,
d'où $\ell'=\inf y_n\le z$, donc $\forall z>\ell$, $\ell'\le z$, d'où $\ell'\le\ell$.

Soit $z<\ell$, alors $\forall n\in\N$, $y_n=\sup_{p\ge n}x_p\ge z$. Comme parmi l'infinité des $x_n>z$, il y en a dont l'indice $p\ge n$ ; alors
$\ell'=\inf y_n\ge z$, donc $\forall z<\ell$, $\ell'\ge z$, d'où $\ell'\ge\ell$.

\begin{note}
La preuve du point 2) (limite inférieure) n'est pas écrite dans le manuscrit.
\end{note}

\begin{enonce}{Proposition}{21}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item $\limsup x_n$ est la plus grande valeur d'adhérence de la suite $(x_n)$.
  \item $\liminf x_n$ est la plus petite valeur d'adhérence de la suite $(x_n)$.
\end{enumerate}
\end{enonce}

\textbf{\color{bleu}Rappel.}
\begin{enumerate}[label=\textbf{\arabic*)},leftmargin=1.2cm]
  \item $a$ est une valeur d'adhérence (ou point d'accumulation) de la suite $(x_n)$ si
  $\forall\varepsilon>0$, il existe une infinité de termes de la suite tels que $|x_n-a|<\varepsilon$, ou de façon équivalente
  $\forall\varepsilon>0$, $\forall N\in\N$, $\exists n\in\N$, $n>N$ et $|x_n-a|<\varepsilon$.
  \item Ainsi $a$ est valeur d'adhérence de $(x_n)$ si et seulement si $\forall\varepsilon>0$, $\forall N\in\N$, $\exists n\ge N$ tel que
  $x_n\in]a-\varepsilon,a+\varepsilon[$ et $x_n\neq a$
  \[
    \iff\ \forall\varepsilon>0,\ \forall N\in\N,\ ]a-\varepsilon,a+\varepsilon[\,\cap A_N\neq\varnothing
    \iff a\in\overline{A_N},\ \forall N\in\N
    \iff a\in\bigcap_{n\in\N}\overline{A_n},\quad A_n=\{x_n,x_{n+1},\dots\}.
  \]
\end{enumerate}

\begin{preuve}\end{preuve}
\textbf{1) Cas de $\limsup x_n$.} Posons $A=\bigcap_{n\in\N}\overline{A_n}$, l'ensemble des valeurs d'adhérence de la suite $(x_n)_{n\in\N}$.
Montrons que $\limsup x_n\in A$ et que $\limsup x_n=\sup A$.

\emph{Remarque :} si $B\subset\R$, alors $\sup B\in\overline B$ (s'il existe). En effet, $\sup B$ est un majorant de $B$ et $\sup B\ge a$ $\forall a\in B$.
De plus $\forall\varepsilon>0$, $\exists a_0\in B$ tel que $a_0>\sup B-\varepsilon$ ; ainsi $\forall\varepsilon>0$, $\exists a_0\in B$ tel que
$\sup B-\varepsilon\le a_0\le\sup B+\varepsilon$, donc $]\sup B-\varepsilon,\sup B+\varepsilon[\,\cap B\neq\varnothing$ $\forall\varepsilon>0$, d'où $\sup B\in\overline B$.

Posons $y_n=\sup_{p\ge n}x_p=\sup A_n$, donc $y_n\in\overline{A_n}$. Or $A_{n+1}\subset A_n$, donc $\overline{A_{n+1}}\subset\overline{A_n}$, ainsi
$\forall m\ge n$, $y_m\in\overline{A_m}\subset\overline{A_n}$, donc $\ell=\inf y_n\in\overline{A_n}$, d'où $\forall n\in\N$, $\ell\in\overline{A_n}$,
c'est-à-dire $\ell\in\bigcap_{n\in\N}\overline{A_n}=A$.

Donc $\limsup x_n$ est une valeur d'adhérence.

Il reste à montrer que $\ell=\limsup x_n=\sup A$. $\forall n\in\N$, on a $\overline{A_n}=[-\infty,y_n]$, donc
\[
  A=\bigcap_{n\in\N}\overline{A_n}=\bigcap_{n\in\N}[-\infty,y_n]=[-\infty,\ell],
\]
donc $\ell=\sup A$.

\begin{note}
La preuve du point 2) (valeur d'adhérence la plus petite pour $\liminf$) n'est pas écrite dans le manuscrit.
\end{note}

\section*{Limites de suites dans $\Rb$}

\begin{enonce}{Théorème}{22}
Pour que la suite $(x_n)_n$ dans $\Rb$ ait une limite $\ell$, il faut et il suffit que
\[
  \limsup x_n \;=\; \liminf x_n \;=\; \ell .
\]
\end{enonce}

\begin{preuve}\end{preuve}

\textbf{1-} \textit{Hypothèse :} la suite $(x_n)_n$ converge vers $\ell$.
$\Rb$ étant séparé, il n'y a qu'une unique valeur d'adhérence $\ell$,
donc $\limsup x_n=\ell$ et $\liminf x_n=\ell$.

\textbf{2-} \textit{Hypothèse :} $\limsup x_n=\liminf x_n=\ell$.
La suite $(x_n)_n$ a donc une seule valeur d'adhérence $\ell$.

Montrons que $(x_n)_n$ converge vers $\ell$.

Supposons que la suite $(x_n)_n$ ne tend pas vers $\ell$.
Alors il existe $V$ voisinage de $\ell$ tel que
\[
  \forall N\in\N,\ \exists\, n>N \text{ et } x_n\notin V .
\]
On peut donc former une sous-suite $(x_{n_i})$ telle que $x_{n_i}\notin V$. En effet :
\begin{itemize}[leftmargin=2cm,label={},itemsep=2pt]
  \item Pour $N=1$, $\exists\, n_1>1$ tel que $x_{n_1}\notin V$.
  \item Pour $N=n_1$, $\exists\, n_2>n_1$ tel que $x_{n_2}\notin V$.
  \item[\hspace{0.3cm}$\vdots$]
  \item Pour $N=n_{i-1}$, $\exists\, n_i>n_{i-1}$ tel que $x_{n_i}\notin V$.
\end{itemize}
La suite $(x_{n_i})_{i\in\N}$ dans $\Rb$ admet au moins une valeur d'adhérence $\ell'$
qui ne peut être $\ell$ puisque les $x_{n_i}\notin V$. D'où la contradiction.

\begin{note}
Pour que $\ell'\neq\ell$, il faut choisir $V$ \emph{ouvert} : une valeur d'adhérence d'une suite
qui reste hors d'un ouvert contenant $\ell$ ne peut pas être $\ell$.
\end{note}

\section*{Rappels de définitions sur les fonctions numériques}

Soit $f,g : E\to\R$ deux applications. On définit :
\begin{align*}
  f+g &: E\to\R, & x&\longmapsto (f+g)(x)=f(x)+g(x),\\
  f\cdot g &: E\to\R, & x&\longmapsto (f\cdot g)(x)=f(x)\,g(x),\\
  \sup(f,g) &: E\to\R, & x&\longmapsto \sup(f,g)(x)=\sup\bigl(f(x),g(x)\bigr),\\
  \inf(f,g) &: E\to\R, & x&\longmapsto \inf(f,g)(x)=\inf\bigl(f(x),g(x)\bigr).
\end{align*}
\[
  f^{+}:E\to\R,\quad x\longmapsto f^{+}(x)=
  \begin{cases} f(x) & \text{si } f(x)\ge 0\\ 0 & \text{si } f(x)\le 0\end{cases}
  \;=\;\sup(f,0)
\]
\[
  f^{-}:E\to\R,\quad x\longmapsto f^{-}(x)=
  \begin{cases} 0 & \text{si } f(x)\ge 0\\ -f(x) & \text{si } f(x)\le 0\end{cases}
  \;=\;\sup(-f,0)
\]
Et on a : $f=f^{+}-f^{-}$.

Les définitions sont les mêmes si $f,g : E\to\Rb$.
Mais $f+g$ n'est pas définie s'il existe un $x\in E$ tel que $f(x)=+\infty$ et $g(x)=-\infty$.

\begin{note}
Même problème si $f(x)=-\infty$ et $g(x)=+\infty$ : les deux cas sont exclus (voir Théorème 24, point 2).
\end{note}

Soit $(f_i)_{i\in I}: E\to\Rb$ une famille quelconque de fonctions. On pose :
\begin{align*}
  \sup_{i\in I} f_i &: E\to\Rb, & x&\longmapsto \Bigl(\sup_{i\in I} f_i\Bigr)(x)=\sup_{i\in I}\bigl(f_i(x)\bigr),\\
  \inf_{i\in I} f_i &: E\to\Rb, & x&\longmapsto \Bigl(\inf_{i\in I} f_i\Bigr)(x)=\inf_{i\in I}\bigl(f_i(x)\bigr),
\end{align*}
qui existent toujours car $\Rb$ est borné.

Si $f_n:E\to\Rb$ est une suite de fonctions, on définit :
\[
  \limsup f_n=\inf_{n\in\N}\Bigl(\sup_{p\ge n} f_p\Bigr)
  \qquad\text{et}\qquad
  \liminf f_n=\sup_{n\in\N}\Bigl(\inf_{p\ge n} f_p\Bigr),
\]
avec $\limsup f_n(x)=\limsup\bigl(f_n(x)\bigr)$.

$\limsup f_n=\liminf f_n$ équivaut à dire que $(f_n)$ converge simplement.

\begin{note}
La définition de $\liminf$ est peu lisible dans le manuscrit ; j'ai rétabli la définition usuelle
$\sup_n\inf_{p\ge n}$.
\end{note}

\section*{2-3) Fonctions numériques mesurables}

\begin{enonce}{Proposition}{23}
Soit $(E,\A)$ un espace mesurable. Soit $f,g:E\to\R$ (ou $\Rb$) deux applications mesurables,
alors l'application
\[
  h:E\to\R\times\R,\qquad x\longmapsto h(x)=\bigl(f(x),g(x)\bigr)
\]
est mesurable de $(E,\A)$ dans $(\R\times\R,\B_{\R\times\R})$ (ou $(\Rb\times\Rb,\B_{\Rb\times\Rb})$).
\end{enonce}

\begin{preuve}\end{preuve}

\begin{center}
\begin{tikzpicture}[>=Stealth,thick,node distance=5cm]
  \node (E)  at (0,3) {$E$};
  \node (RR) at (7,3) {$\Rb\times\Rb$};
  \node (R)  at (7,0) {$\Rb$};
  \draw[->] (E) -- node[above]{$h$} node[below]{\small$x\mapsto(f(x),g(x))$} (RR);
  \draw[->] (RR) -- node[right]{$p$} (R);
  \draw[->] (E) -- node[below left,sloped]{$f\cdot g=p\circ h$} (R);
  \node[right=2pt of R,xshift=0.2cm,font=\small] {$f(x)\cdot g(x)$};
\end{tikzpicture}
\end{center}

Soit $x\in E$. Alors
\[
  (p\circ h)(x)=p\bigl(h(x)\bigr)=p\bigl(f(x),g(x)\bigr)=f(x)\cdot g(x)=(f\cdot g)(x).
\]
Dans le cas de $\R\times\R$, comme les pavés $[a,b]\times[c,d]$ engendrent $\B_{\R\times\R}$,
il suffit de montrer que
\[
  h^{-1}\bigl([a,b]\times[c,d]\bigr)\in\A .
\]
Or :
\begin{align*}
  h^{-1}\bigl([a,b]\times[c,d]\bigr)
    &=\bigl\{x\in E : h(x)\in[a,b]\times[c,d]\bigr\}\\
    &=\bigl\{x\in E : f(x)\in[a,b]\ \text{et}\ g(x)\in[c,d]\bigr\}\\
    &=f^{-1}\bigl([a,b]\bigr)\cap g^{-1}\bigl([c,d]\bigr)\in\A
\end{align*}
car $f$ et $g$ sont mesurables et $\A$ est une tribu.

%---------------------------------------------------------------------
\begin{enonce}{Théorème}{24}
Soit $(E,\A)$ un espace mesurable. Soit $f,g:E\to\Rb$ mesurables, alors
$f+g$, $f\cdot g$, $\alpha f\ (\alpha\in\R)$, $\sup(f,g)$, $\inf(f,g)$, $f^{+}$ et $f^{-}$ sont mesurables.
\end{enonce}

\begin{preuve}\end{preuve}

\textbf{1- Pour $f\cdot g : E\to\Rb$.} $f\cdot g$ est la composée de deux fonctions suivantes :
\[
  h:E\to\Rb\times\Rb,\ \ x\mapsto h(x)=\bigl(f(x),g(x)\bigr)
  \qquad\text{et}\qquad
  p:\Rb\times\Rb\to\Rb,\ \ (u,v)\mapsto u\cdot v .
\]
$h$ et $p$ étant mesurables, alors $f\cdot g=p\circ h$ est mesurable.

\textbf{2- Pour $f+g$ :} utiliser l'application de $E\to\Rb\times\Rb\setminus\{(-\infty,+\infty),(+\infty,-\infty)\}$.

\begin{note}
La preuve est seulement esquissée dans le manuscrit : on compose $h$ avec l'addition $(u,v)\mapsto u+v$,
mesurable sur $\Rb\times\Rb$ privé des deux couples ci-dessus.
\end{note}

\textbf{3- Pour $\sup(f,g)$.} $\sup(f,g)$ est la composée de :
\[
  h:E\to\Rb\times\Rb,\ \ x\mapsto h(x)=\bigl(f(x),g(x)\bigr)
  \qquad\text{et}\qquad
  S:\Rb\times\Rb\to\Rb,\ \ (u,v)\mapsto\sup(u,v).
\]
Donc $\sup(f,g)=S\circ h$. $h$ étant mesurable, il reste à montrer que
$S:(\Rb\times\Rb,\B_{\Rb\times\Rb})\to(\Rb,\B_{\Rb})$ est mesurable.

Or la famille des $\bigl([-\infty,a],\ a\in\R\bigr)$ engendre $\B_{\Rb}$, il suffit donc de montrer que
les $S^{-1}([-\infty,a])$ sont des mesurables (appartiennent à $\B_{\Rb\times\Rb}$). Or :
\[
  S^{-1}\bigl([-\infty,a]\bigr)=\bigl\{(x,y)\in\Rb\times\Rb : S(x,y)\in[-\infty,a]\bigr\}
  =[-\infty,a]\times[-\infty,a]\in\B_{\Rb\times\Rb}.
\]
Ainsi, $\sup(f,g)$ est mesurable.

\textbf{4- Pour $f^{+}$.} On a $f^{+}=\sup(f,0)$. Comme $f$ est mesurable et $0$ est mesurable,
alors $f^{+}$ est mesurable.

\begin{note}
Les cas $\inf(f,g)$, $\alpha f$ et $f^{-}$ ne sont pas détaillés dans le manuscrit.
Ils se traitent de même : $\inf(f,g)=-\sup(-f,-g)$ et $f^{-}=\sup(-f,0)$.
\end{note}

%---------------------------------------------------------------------
\begin{enonce}{Théorème}{25}
Soit $(f_n)$ une suite de fonctions numériques mesurables à valeurs dans $\Rb$, alors :
\begin{enumerate}[label=\textbf{\arabic*-},leftmargin=1.2cm]
  \item Les fonctions $\sup f_n$, $\inf f_n$, $\limsup f_n$ et $\liminf f_n$ sont mesurables.
  \item Si la suite $f_n$ converge simplement vers $f$, alors $f$ est mesurable.
\end{enumerate}
\end{enonce}

\begin{preuve}\end{preuve}

\textbf{1-} On a $\displaystyle\limsup f_n=\inf_{n\in\N}\Bigl(\sup_{p\ge n}f_p\Bigr)$.
On montre que $\limsup f_n$ est mesurable lorsqu'on a au préalable montré que
$\sup f_n$ et $\inf f_n$ sont mesurables.

Posons $g=\sup f_n : (E,\A)\to(\Rb,\B_{\Rb})$ avec $f_n$ mesurable.
Comme les $[-\infty,a]$ engendrent $\B_{\Rb}$, il suffit de montrer que
\[
  \forall a\in\R,\quad g^{-1}\bigl([-\infty,a]\bigr)\in\A .
\]
Soit $a\in\R$, alors :
\[
  g^{-1}\bigl([-\infty,a]\bigr)=\bigl\{x\in E : g(x)\in[-\infty,a]\bigr\}
  =\bigl\{x\in E : \sup_n f_n(x)\le a\bigr\}
  =\bigcap_{n\in\N} f_n^{-1}\bigl([-\infty,a]\bigr)\in\A
\]
car $f_n$ est mesurable et $\A$ est une tribu.

\begin{enonce}{Remarque}{26}
Ceci marche pour le $\sup$ ou le $\inf$ d'une famille dénombrable.
\end{enonce}

\begin{note}
Le manuscrit s'arrête ici : les preuves pour $\inf f_n$, $\limsup f_n$, $\liminf f_n$
et le point 2 du Théorème 25 ne sont pas écrites.
\end{note}

\section*{2-4) Fonctions étagées mesurables}

\begin{enonce}{Définition}{27}
\begin{enumerate}[label=\textbf{\arabic*-},leftmargin=1.2cm]
  \item Une application $f:E\to F$ est dite \emph{étagée} si elle prend un nombre fini de valeurs
        $(y_1,y_2,\dots,y_n)$ de $F$. Donc si $f(E)=\{y_1,y_2,\dots,y_n\}$.
  \item Si $\A$ est une famille de parties de $E$ ($\A\subset\Pp(E)$), $f:E\to F$ est dite
        \emph{étagée sur $\A$} si elle est étagée et si pour toute valeur $y_i\in F$ prise par $f$,
        $f^{-1}(\{y_i\})\in\A$.
\end{enumerate}
\end{enonce}

\begin{enonce}{Proposition}{28}
Pour qu'une application étagée $f:(E,\A)\to(\R,\B_{\R})$ soit mesurable,
il faut et il suffit qu'elle soit étagée sur $\A$.
\end{enonce}

\begin{preuve}\end{preuve}

($\Rightarrow$) \textit{Hypothèse :} $f$ est étagée et mesurable. Montrons que $f$ est étagée sur $\A$.
Soit $y_i\in\R$, alors $\{y_i\}$ est un borélien ; $f$ étant mesurable, alors $f^{-1}(\{y_i\})\in\A$.
D'où $f$ est étagée sur $\A$.

($\Leftarrow$) \textit{Hypothèse :} $f^{-1}(\{y_i\})\in\A$ pour $i=1,2,\dots,n$.
Montrons que $f$ est mesurable. Soit $A\in\B_{\R}$, a-t-on $f^{-1}(A)\in\A$ ? On a :
\[
  f^{-1}(A)=\bigcup_{y_i\in A} f^{-1}\bigl(\{y_i\}\bigr)\in\A
\]
car réunion finie d'éléments de $\A$, où $A=\bigcup_{y_i\in A}\{y_i\}$.

\begin{note}
Plus précisément, la réunion porte sur les $y_i\in A\cap f(E)$ : $f^{-1}(A)=f^{-1}(A\cap f(E))$.
Le symbole $A$ désigne ici un borélien et ailleurs la tribu $\A$ (notations proches dans le manuscrit).
\end{note}

%---------------------------------------------------------------------
\begin{enonce}{Proposition}{29}
L'ensemble des fonctions étagées sur la $\sigma$-algèbre $\A$ et à valeurs dans $\R$
est un espace vectoriel $\Ee(\A,\R)$.
\end{enonce}

\begin{preuve}\end{preuve}

Soit $f(E)=\{y_1,y_2,\dots,y_n\}$ et $g(E)=\{z_1,z_2,\dots,z_p\}$.
Alors $f+g$ prend des valeurs dans l'ensemble fini $\{y_i+z_j\}_{\,i=1,\dots,n;\ j=1,\dots,p}$.

Soit $x$ une valeur prise par $f+g$ ; on veut montrer que $(f+g)^{-1}(x)\in\A$.
Il y a un nombre fini de couples $(i,j)$ tels que $y_i+z_j=x$, et
$f^{-1}(\{y_i\})\cap g^{-1}(\{z_j\})\in\A$ car $f$ et $g$ sont étagées sur $\A$ et $\A$ est une tribu.
Ainsi :
\[
  \bigcup_{(i,j)\in I}\Bigl(f^{-1}(\{y_i\})\cap g^{-1}(\{z_j\})\Bigr)\in\A,
  \qquad\text{où } I=\bigl\{(i,j) : y_i+z_j=x\bigr\}.
\]
D'où $(f+g)^{-1}(x)\in\A$.

\begin{note}
Seule la stabilité par addition est écrite dans le manuscrit ;
la stabilité par multiplication par un scalaire ($\lambda f$) se traite de la même façon.
\end{note}

\begin{enonce}{Théorème}{30}
Soit $(E,\A)$ un espace mesurable. Toute fonction mesurable positive (à valeurs dans $\overline{\R_+}$)
est limite d'une suite croissante de fonctions étagées positives sur $\A$.
\end{enonce}

\begin{preuve}\end{preuve}

Soit $f:E\to\overline{\R_+}$ mesurable. Construisons une suite $(f_n)$ croissante de fonctions étagées
et positives qui tend vers $f$.

\begin{center}
\begin{tikzpicture}[scale=1,>=Stealth]
  % bande [k/2^n,(k+1)/2^n[
  \fill[bleuclair!15] (-0.2,2.2) rectangle (11.2,3.6);
  \foreach \y in {0.9,2.2,3.6,4.9}{\draw[gray!70,thin] (-0.2,\y) -- (11.2,\y);}
  % axes
  \draw[->,thick] (-0.2,0) -- (11.4,0);
  \draw[->,thick] (0,-0.5) -- (0,5.6);
  % graphe de f
  \draw[thick,red!60!black,smooth] plot coordinates
    {(0.3,1.4)(1.0,1.9)(1.8,1.6)(2.6,0.9)(3.4,1.0)(4.2,1.7)(4.9,2.9)(5.7,3.4)(6.6,3.0)(7.4,3.2)(8.2,3.5)(9.0,2.6)(9.8,1.4)(10.6,1.1)};
  % étiquettes
  \node[left] at (0,2.2) {$\dfrac{k}{2^n}$};
  \node[left] at (0,3.6) {$\dfrac{k+1}{2^n}$};
  \node[red!60!black] at (10.8,4.2) {$f(x)$};
  % Q_{n,k}
  \draw[bleuclair,line width=2.5pt] (4.62,-0.18) -- (8.75,-0.18);
  \draw[bleuclair,dotted] (4.62,-0.18) -- (4.62,2.2);
  \draw[bleuclair,dotted] (8.75,-0.18) -- (8.75,2.2);
  \node[bleuclair,below] at (6.7,-0.3) {$Q_{n,k}=f^{-1}\!\left(\left[\frac{k}{2^n},\frac{k+1}{2^n}\right[\right)$};
\end{tikzpicture}
\end{center}

On construit $(f_n)$ de la manière suivante.
\begin{itemize}[leftmargin=1.2cm]
  \item Dans $R_n=f^{-1}\bigl([n,+\infty]\bigr)$, on pose : $f_n(x)=n$.
  \item Dans chaque $Q_{n,k}=f^{-1}\Bigl(\Bigl[\dfrac{k}{2^n},\dfrac{k+1}{2^n}\Bigr[\Bigr)$, on pose :
        $f_n(x)=\dfrac{k}{2^n}$ \quad (lorsque $\dfrac{k}{2^n}\le f(x)<\dfrac{k+1}{2^n}$).
\end{itemize}
Alors $0\le k\le n\times 2^n$.

\begin{note}
Le manuscrit écrit « $f_n(x)=n$ ($x\ge n$) » et « $\frac{k}{2^n}\le x<\frac{k+1}{2^n}$ » :
c'est en réalité $f(x)$ qui intervient (et non $x$). De plus, avec ces intervalles semi-ouverts,
on a $0\le k\le n\,2^n-1$.
\end{note}

\textbf{1-} Les premiers points sont étagés sur $\A$ car $Q_{n,k}$ et $R_n\in\A$.

\textbf{2-} Montrons que la suite $(f_n)$ est croissante. Il suffit de montrer que dans chaque domaine
$Q_{n,k}$, $R_n$ on a : $f_n(x)\le f_{n+1}(x)$.

On a :
\[
  \Bigl[\frac{k}{2^n},\frac{k+1}{2^n}\Bigr[
  =\Bigl[\frac{2k}{2^{n+1}},\frac{2k+1}{2^{n+1}}\Bigr[\ \cup\ \Bigl[\frac{2k+1}{2^{n+1}},\frac{2k+2}{2^{n+1}}\Bigr[
\]
Donc $Q_{n,k}=Q_{n+1,2k}\cup Q_{n+1,2k+1}$.

Si $x\in R_n$ : $[n,+\infty]=[n,n+1[\ \cup\ [n+1,+\infty]$, soit
\[
  [n,+\infty]=\Biggl(\bigcup_{n\times 2^{n+1}\,\le\,k\,\le\,(n+1)2^{n+1}}
  \Bigl[\frac{k}{2^{n+1}},\frac{k+1}{2^{n+1}}\Bigr[\Biggr)\ \cup\ [n+1,+\infty].
\]
Ainsi :
\[
  R_n=\Biggl(\bigcup_{n\times 2^{n+1}\,\le\,k\,\le\,(n+1)2^{n+1}} Q_{n+1,k}\Biggr)\cup R_{n+1},
\]
où, selon le cas :
\begin{itemize}[leftmargin=1.2cm]
  \item $f_{n+1}(x)=\dfrac{k}{2^{n+1}}\ \ge\ n=f_n(x)$ ;
  \item ou bien $f_{n+1}(x)=n+1>n=f_n(x)$.
\end{itemize}
Donc $f_n$ est croissante.

\begin{note}
Dans la réunion, $k$ doit aller de $n\,2^{n+1}$ à $(n+1)\,2^{n+1}-1$ pour que les $Q_{n+1,k}$
soient bien contenus dans $R_n$.
\end{note}

\textbf{3-} Montrons que $f_n\to f$ simplement.

\emph{1\textsuperscript{er} cas :} si $f(x)\neq+\infty$, alors $\exists\, n_0\in\N$ tel que $n_0>f(x)$.
Pour $n\ge n_0$, $x\in Q_{n,k}$ tel que
\[
  \frac{k}{2^n}\le f(x)<\frac{k+1}{2^n}\qquad\text{et}\qquad f_n(x)=\frac{k}{2^n},
\]
ainsi $f(x)-f_n(x)<\dfrac{1}{2^n}\xrightarrow[n\to+\infty]{}0$.

\emph{2\textsuperscript{e} cas :} si $f(x)=+\infty$, alors $\forall n$, $x\in R_n$, donc
$f_n(x)=n\xrightarrow[n\to+\infty]{}+\infty=f(x)$.

Ainsi, $f_n(x)\to f(x)$.

\begin{enonce}{Corollaire}{31}
Soit $(E,\A)$ un espace mesurable. Toute fonction numérique mesurable est limite simple
d'une suite de fonctions étagées sur $\A$ à valeurs dans $\R$.
\end{enonce}

\begin{preuve}\end{preuve}

On écrit $f=f^{+}-f^{-}$, et $f^{+}$ et $f^{-}$ sont des fonctions mesurables positives.
On peut leur appliquer le théorème précédent.

Donc il existe une suite croissante $(g_n)$ de fonctions étagées positives sur $\A$ telle que
$g_n\to f^{+}$. De même, il existe une suite croissante $(h_n)$ de fonctions étagées positives
sur $\A$ telle que $h_n\to f^{-}$.

$g_n-h_n$ est étagée sur $\A$ et à valeurs dans $\R$, et
\[
  g_n-h_n\longrightarrow f^{+}-f^{-}=f .
\]

\begin{note}
« Étagée sur $\A$ » résulte de la Proposition 29 (espace vectoriel).
\end{note}
